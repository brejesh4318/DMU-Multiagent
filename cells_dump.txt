# CELL 7 — LLM + LLM-AS-JUDGE
# =========================================================
# LLM-as-Judge: a SEPARATE LLM call that evaluates
# whether the generated SQL and answer are correct.
#
# Why: LLMs can generate plausible-looking but wrong SQL.
# The judge catches this BEFORE showing to District Collector.
# =========================================================

from langchain_groq import ChatGroq

llm   = ChatGroq(model=CFG['LLM_MODEL'],   temperature=CFG['TEMPERATURE'])
judge = ChatGroq(model=CFG['JUDGE_MODEL'],  temperature=0.0)


JUDGE_PROMPT = """You are an expert SQL auditor for educational analytics.
Evaluate whether the SQL and answer correctly answer the user's question.

Question: {question}
SQL: {sql}
SQL Result: {result}
Answer: {answer}

Evaluate on:
1. sql_correct (bool): Does the SQL correctly answer the question?
2. result_reasonable (bool): Does the result make logical sense?
3. answer_grounded (bool): Is the answer based only on the SQL result?
4. score (1-5): Overall quality (5=perfect)
5. issues (list): List specific problems if any
6. suggestion (str): How to fix if score < 4

Known facts for validation:
- All data is Nilgiris district, Tamil Nadu
- Years are 2025 and 2026 only
- Pass mark per subject is 35
- In 2026 data, failures show as NULL in PASS column
- Students take different subjects by group (commerce vs science)

Return ONLY valid JSON, no other text:
{{"sql_correct": bool, "result_reasonable": bool, "answer_grounded": bool,
  "score": int, "issues": [str], "suggestion": str}}"""


def llm_judge(question: str, sql: str, result: str, answer: str) -> dict:
    """Evaluate SQL + answer quality. Returns dict with score and issues."""
    if not CFG['JUDGE_ENABLED']:
        return {'score': 5, 'issues': [], 'suggestion': '', 'sql_correct': True,
                'result_reasonable': True, 'answer_grounded': True}
    try:
        prompt = JUDGE_PROMPT.format(
            question=question, sql=sql[:500],
            result=result[:300], answer=answer[:300]
        )
        resp = judge.invoke(prompt).content.strip()
        # clean markdown fences if LLM adds them
        if '```json' in resp: resp = resp.split('```json')[1].split('```')[0].strip()
        elif '```'   in resp: resp = resp.split('```')[1].split('```')[0].strip()
        return json.loads(resp)
    except Exception as e:
        logger.warning(f'Judge failed: {e}')
        return {'score': 3, 'issues': [str(e)], 'suggestion': 'Judge parse error',
                'sql_correct': True, 'result_reasonable': True, 'answer_grounded': True}

# CELL 8 — SQL ENGINE WITH RETRY + JUDGE LOOP
# =========================================================

def generate_and_execute_sql(question: str) -> dict:
    """
    Full SQL pipeline:
    1. Generate SQL from question
    2. Execute on DuckDB
    3. LLM-as-judge evaluates
    4. If score < threshold: retry with judge feedback
    5. Generate clean natural language answer
    Returns: {answer, sql, result_df, judge, attempts}
    """
    sql, result_df, error = '', None, ''
    judge_feedback = ''
    attempts = 0

    for attempt in range(CFG['SQL_RETRIES'] + 1):
        attempts = attempt + 1

        # build prompt with optional judge feedback
        extra = ''
        if judge_feedback:
            extra = f'\nPREVIOUS ATTEMPT FAILED. Judge feedback: {judge_feedback}\nFix the SQL.\n'

        prompt = SQL_SYSTEM_PROMPT.replace('{question}', question) + extra

        try:
            resp = llm.invoke(prompt).content.strip()
            if '```sql' in resp: resp = resp.split('```sql')[1].split('```')[0].strip()
            elif '```'  in resp: resp = resp.split('```')[1].split('```')[0].strip()
            sql = resp

            result_df = con.execute(sql).fetchdf()
            error     = ''
            logger.info(f'SQL attempt {attempts} succeeded: {sql[:60]}')

        except Exception as e:
            error = str(e)
            judge_feedback = f'SQL execution error: {error}'
            logger.warning(f'SQL attempt {attempts} failed: {error[:80]}')
            time.sleep(0.5)
            continue

        # format result
        if result_df is None or result_df.empty:
            result_str = 'No matching records found.'
        else:
            result_str = result_df.to_string(index=False)

        # generate natural language answer
        answer_prompt = (
            f'Question: {question}\n\n'
            f'Data from database:\n{result_str}\n\n'
            f'Rules:\n'
           
            f'- Use ONLY numbers from the data above\n'
            f'- If data is empty, say so clearly\n'
            f'- Plain text only, no code, no backticks\n'
            f'- Be precise and actionable\nAnswer:'
        )
        answer = llm.invoke(answer_prompt).content.strip()

        # LLM-as-judge evaluation
        j = llm_judge(question, sql, result_str[:300], answer[:300])
        logger.info(f'Judge score: {j["score"]}/5  issues: {j["issues"]}')

        if j['score'] >= CFG['JUDGE_RETRY_THRESHOLD'] or attempt >= CFG['SQL_RETRIES']:
            return {
                'answer':     answer,
                'sql':        sql,
                'result_df':  result_df,
                'result_str': result_str,
                'judge':      j,
                'attempts':   attempts,
                'error':      None,
            }
        else:
            # retry with judge feedback
            judge_feedback = f'Score {j["score"]}/5. Issues: {j["issues"]}. Suggestion: {j["suggestion"]}'
            logger.warning(f'Retrying due to low judge score: {judge_feedback}')
            time.sleep(0.5)

    return {
        'answer':     f'Could not generate a correct answer after {attempts} attempts. Error: {error}',
        'sql':        sql, 'result_df': None, 'result_str': error,
        'judge':      {'score': 0}, 'attempts': attempts, 'error': error
    }
